cmake_minimum_required(VERSION 3.20)
project(EquipmentExportTests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 14)
set(WMV_ROOT "" CACHE PATH "WMV checkout")
set(wxWidgets_ROOT_DIR "${WMV_ROOT}/ThirdParty/wxWidgets33")
set(wxWidgets_LIB_DIR "${wxWidgets_ROOT_DIR}/lib/vc14x_x64_dll")
find_package(wxWidgets REQUIRED COMPONENTS core base)
include(${wxWidgets_USE_FILE})
find_package(Qt5 REQUIRED COMPONENTS Core)
add_executable(equipment-dialog dialog.cpp "${WMV_ROOT}/Source/wowmodelviewer/EquipmentExportDialog.cpp")
# Only loaded-game data is substituted. Compile and exercise the real dialog and events.
target_include_directories(equipment-dialog PRIVATE fixtures "${WMV_ROOT}/Source/wowmodelviewer")
target_link_libraries(equipment-dialog PRIVATE ${wxWidgets_LIBRARIES} Qt5::Core)

set(WMV_FBX_SDK "${WMV_ROOT}/ThirdParty/fbx_x64" CACHE PATH "FBX SDK for independent reimport checks")
if(EXISTS "${WMV_FBX_SDK}/include/fbxsdk.h")
  add_executable(equipment-inspect inspect.cpp)
  target_compile_definitions(equipment-inspect PRIVATE FBXSDK_SHARED)
  target_include_directories(equipment-inspect PRIVATE "${WMV_FBX_SDK}/include")
  target_link_libraries(equipment-inspect PRIVATE "${WMV_FBX_SDK}/lib/libfbxsdk.lib")
endif()
